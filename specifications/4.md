async function refreshDashboardData() {
  // Secured token reference corrected to align with auth.js standard mapping architecture
  const token = localStorage.getItem('urivo_access_token');
  if (!token) {
    window.location.href = '/login';
    return;
  }

  try {
    // 1. Resolve customer identity profiles and dynamic token verification metrics
    const authRes = await fetch('/api/auth/verify-session', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ token })
    });
    
    if (!authRes.ok) throw new Error("Merchant authentication session expired.");
    const authData = await authRes.json();
    
    // UI Hydration Layer for credits and matching user data fields
    document.getElementById('merchantEmail').innerText = authData.user.email;
    document.getElementById('creditDisplay').innerHTML = `
      ${authData.user.credits_remaining} <span class="text-xs font-sans tracking-wide text-urivo-canvas/60">Credits</span>

    const response = await fetch(`/api/merchant/stores`, {
      method: 'GET',

    console.error("BEOS Dashboard Sync Exception:", error);
  }
}

function escapeHTML(str) {
  if (!str) return '';
  return str.replace(/&/g, '&amp;')
            .replace(/</g, '&lt;')
            .replace(/>/g, '&gt;')
            .replace(/"/g, '&quot;')
            .replace(/'/g, '&#x27;')
            .replace(/\//g, '&#x2F;');
}

// 2. TOKEN OWNSERSHIP SECURITY LAYER MIDDLEWARE
async function enforceTenantIsolation(req, res, next) {
  const authHeader = req.headers.authorization;
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ error: "Access token missing or malformed." });
  }
  try {
    const token = authHeader.split(' ')[1];
    const { data: { user }, error } = await supabase.auth.getUser(token);
    if (error || !user) return res.status(401).json({ error: "Authentication checkpoint verification failed." });
    
    req.authenticatedMerchantId = user.id;
    next();
  } catch (err) {
    return res.status(401).json({ error: "Critical session validation failure." });
  }
}
