/**
 * URIVO Layout Realignment Subsystem
 * Architectural Standard: BEOS Book II, III & VI
 * Description: Fully authenticated configuration updates checking row-level ownership patterns.
 */
const express = require('express');
const router = express.Router();
const { supabase } = require('../config/supabase');

router.put('/api/stores/update-layout', async (req, res) => {
  const { storeId, storeName, tagline, palette } = req.body;
  const authHeader = req.headers.authorization;

  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ error: "Access verification token signature missing." });
  }

  if (!storeId || !storeName || !tagline || !palette) {
    return res.status(400).json({ error: "Incomplete configuration payload parameters received." });
  }

  try {
    const token = authHeader.split(' ')[1];
    
    // 1. Resolve security profile identifier context using GoTrue verification
    const { data: { user }, error: authError } = await supabase.auth.getUser(token);
    if (authError || !user) {
      return res.status(401).json({ error: "Merchant identity session signature is invalid or expired." });
    }

    // 2. Fetch original context and verify tenancy ownership criteria
    const { data: activeStore, error: fetchError } = await supabase
      .from('stores')
      .select('user_id, theme_config')
      .eq('id', storeId)
      .single();

    if (fetchError || !activeStore) {
      return res.status(104).json({ error: "Target storefront node context not located." });
    }

    // Explicit Ownership Gate to prevent multi-tenant cross-spoofing
    if (activeStore.user_id !== user.id) {
      return res.status(403).json({ error: "Operation rejected. Insufficient security clearance privileges." });
    }

    // 3. Assemble and merge structural payload updates safely
    const updatedThemeConfig = {
      ...activeStore.theme_config,
      tagline: tagline.trim(),
      palette: {
        ...activeStore.theme_config.palette,
        primaryBackground: palette.primaryBackground.trim(),
        secondaryStructure: palette.secondaryStructure.trim()
      }
    };

    // 4. Persist modification maps to the Postgres architecture
    const { data: updatedStore, error: updateError } = await supabase
      .from('stores')
      .update({
        store_name: storeName.trim(),
        theme_config: updatedThemeConfig
      })
      .eq('id', storeId)
      .select()
      .single();

    if (updateError) throw updateError;

    return res.status(200).json({
      success: true,
      message: "Storefront visual schema updated cleanly.",
      store: updatedStore
    });

  } catch (error) {
    console.error("BEOS Subsystem Failure [update-layout]:", error);
    return res.status(500).json({ error: "Failed to persist committed design changes." });
  }
});

module.exports = router;

/**
 * URIVO Inventory Infrastructure Router
 * Architectural Standard: BEOS Book II & VI
 * Description: Coordinates item lifecycle additions/deletions under strong tenancy isolation boundaries.
 */
const express = require('express');
const router = express.Router();
const { supabase } = require('../config/supabase');

// 1. Secure Item Deployment
router.post('/api/products/create', async (req, res) => {
  const { storeId, title, description, priceEur, inventoryCount } = req.body;
  const authHeader = req.headers.authorization;

  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ error: "Access verification token signature missing." });
  }

  if (!storeId || !title || !description || isNaN(priceEur) || isNaN(inventoryCount)) {
    return res.status(400).json({ error: "Incomplete item parameters received." });
  }

  try {
    const token = authHeader.split(' ')[1];
    const { data: { user }, error: authError } = await supabase.auth.getUser(token);
    if (authError || !user) return res.status(401).json({ error: "Invalid merchant authorization signatures." });

    // Enforce tenancy verify verification maps
    const { data: store, error: storeError } = await supabase
      .from('stores')
      .select('user_id')
      .eq('id', storeId)
      .single();

    if (storeError || !store || store.user_id !== user.id) {
      return res.status(403).json({ error: "Write privileges denied for targeted storefront vector." });
    }

    const { data: newProduct, error: insertError } = await supabase
      .from('products')
      .insert([{
        store_id: storeId,
        title: title.trim(),
        description: description.trim(),
        price_eur: parseFloat(priceEur),
        inventory_count: parseInt(inventoryCount)
      }])
      .select()
      .single();

    if (insertError) throw insertError;

    return res.status(201).json({ success: true, product: newProduct });

  } catch (error) {
    console.error("BEOS Subsystem Failure [product-create]:", error);
    return res.status(500).json({ error: "Failed to deploy new catalog entity." });
  }
});

// 2. Secure Item Removal Array Purge Execution
router.delete('/api/products/delete', async (req, res) => {
  const { productId } = req.body;
  const authHeader = req.headers.authorization;

  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ error: "Access verification token signature missing." });
  }

  if (!productId) return res.status(400).json({ error: "Missing mandatory parameter: productId." });

  try {
    const token = authHeader.split(' ')[1];
    const { data: { user }, error: authError } = await supabase.auth.getUser(token);
    if (authError || !user) return res.status(401).json({ error: "Expired authorization credentials map signature." });

    // Resolve structural parenting through joints to parse absolute row metrics permissions
    const { data: product, error: productError } = await supabase
      .from('products')
      .select('store_id, stores(user_id)')
      .eq('id', productId)
      .single();

    if (productError || !product || product.stores.user_id !== user.id) {
      return res.status(403).json({ error: "Destructive operations denied for targeted object configuration." });
    }

    const { error: deleteError } = await supabase.from('products').delete().eq('id', productId);
    if (deleteError) throw deleteError;

    return res.status(200).json({ success: true, message: "Catalog entity successfully purged from database layer." });

  } catch (error) {
    console.error("BEOS Subsystem Failure [product-delete]:", error);
    return res.status(500).json({ error: "Failed to securely drop target entity row." });
  }

let currentActiveStoreId = null;

async function executeStorefrontGeneration() {
  const userPromptInput = prompt("Enter your high-end brand concept (e.g., 'Bespoke Scandinavian watch line'):");
  const desiredSubdomainInput = prompt("Enter your desired subdomain (e.g., 'nordicprecision'):");
  
  // Synchronized Storage Keys matching the central session architecture definitions
  const sessionToken = localStorage.getItem('urivo_access_token');
  if (!sessionToken) {
    window.location.href = '/login';
    return;
  }

  if (!userPromptInput || !desiredSubdomainInput) {
    alert("Generation parameters cannot be empty.");
    return;
  }

  const progressInterval = startLoaderSequencer();

  try {
    const userRes = await fetch('/api/auth/verify-session', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ token: sessionToken })
    });
    const userData = await userRes.json();

    if (!userRes.ok) throw new Error("Authorization credentials expired.");

    const response = await fetch('/api/generate-store', {
      method: 'POST',
      headers: { 
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${sessionToken}`
      },
      body: JSON.stringify({
        userId: userData.user.id,
        prompt: userPromptInput,
        desiredSubdomain: desiredSubdomainInput
      })
    });

    const data = await response.json();
    if (!response.ok) throw new Error(data.error || "Generation loop tracking error.");

    terminateLoaderSequencer(progressInterval);
    alert(`Success, bro! Your store is now running live at: ${data.storeUrl}`);
    
    if (typeof refreshDashboardData === 'function') refreshDashboardData();

  } catch (error) {
    terminateLoaderSequencer(progressInterval);
    console.error("BEOS Generation Interaction Failure:", error);
    alert(error.message || "Failed to finalize store configuration matrix.");
  }
}

document.addEventListener('DOMContentLoaded', () => {
  const genBtn = document.querySelector('button');
  if (genBtn && genBtn.innerText.includes('Generate')) {
    genBtn.onclick = executeStorefrontGeneration;
  }
