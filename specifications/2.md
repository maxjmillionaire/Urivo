### FILE 12: services/stripeCheckout.js
// Stripe Subscription Portal Linker: Securely issues validated checkout pipelines.

const express = require('express');
const router = express.Router();
const stripe = require('stripe')(process.env.STRIPE_SECRET_KEY);

router.post('/api/checkout/create-session', async (req, res) => {
  const { priceId, customerEmail } = req.body;
  const baseUrl = process.env.PUBLIC_URL || 'https://urivo.ai';

  if (!priceId) {
    return res.status(400).json({ error: "Missing required billing parameter 'priceId'." });
  }

  try {
    // Generate secure hosted payment instance via Stripe Core API
    const session = await stripe.checkout.sessions.create({
      mode: 'subscription',
      payment_method_types: ['card'],
      customer_email: customerEmail || undefined,
      line_items: [{ price: priceId, quantity: 1 }],
      
      // Fixed structural path parameters for absolute redirect execution
      success_url: `${baseUrl}/dashboard?session_id={CHECKOUT_SESSION_ID}&status=success`,
      cancel_url: `${baseUrl}/dashboard?status=cancelled`,
    });

    return res.status(200).json({ id: session.id, url: session.url });

  } catch (error) {
    console.error("Stripe Checkout Framework Failure:", error);
    return res.status(500).json({ error: "Failed to establish a secure transaction session channel." });
  }
});

module.exports = router;

### FILE 13: services/authRouter.js
// Supabase-Unified Router: Safely registers and signs in merchants using standard GoTrue frameworks.

const express = require('express');
const router = express.Router();
const { supabase } = require('../config/supabase');

// 1. Merchant Registration Gateway
router.post('/api/auth/register', async (req, res) => {
  const { email, password } = req.body;

  if (!email || !password) {
    return res.status(400).json({ error: "Email and password are required parameters." });
  }

  try {
    // Execute registration directly into the central, secure Supabase Auth table
    const { data, error } = await supabase.auth.signUp({
      email: email.toLowerCase().trim(),
      password: password
    });

    if (error) throw error;

    return res.status(201).json({ 
      success: true, 
      message: "Identity configuration initialized. Please confirm email if applicable." 
    });

  } catch (error) {
    console.error("Supabase Auth Infrastructure Failure [Register]:", error.message);
    return res.status(400).json({ error: error.message || "Internal registration lifecycle failure." });
  }
});

// 2. Merchant Login & Dynamic Session Token Issuance
router.post('/api/auth/login', async (req, res) => {
  const { email, password } = req.body;

  try {
    const { data, error } = await supabase.auth.signInWithPassword({
      email: email.toLowerCase().trim(),
      password: password
    });

    if (error || !data.user) {
      return res.status(401).json({ error: "Invalid authentication credentials verification state." });
    }

    // Extract access tokens safely generated by Supabase security core
    return res.status(200).json({
      success: true,
      access_token: data.session.access_token,
      user: {
        id: data.user.id,
        email: data.user.email
      }
    });

  } catch (error) {
    console.error("Supabase Auth Infrastructure Failure [Login]:", error.message);
    return res.status(500).json({ error: "Internal authentication verification process failed." });
  }
});

// 3. Token-to-Profile Synced Verification Layer
router.post('/api/auth/verify-session', async (req, res) => {
  const { token } = req.body;

  if (!token) return res.status(401).json({ error: "Active security session token missing." });

  try {
    // 1. Authenticate access token validity with GoTrue Core
    const { data: { user }, error: authError } = await supabase.auth.getUser(token);
    if (authError || !user) throw new Error("Invalid access signature.");

    // 2. Fetch live data metrics from the public metadata user profile
    const { data: profile, error: profileError } = await supabase
      .from('profiles')
      .select('id, email, credits_remaining, subscription_status')
      .eq('id', user.id)
      .single();

    if (profileError || !profile) throw new Error("Profile synchronization mapping error.");

    return res.status(200).json({ success: true, user: profile });

  } catch (error) {
    return res.status(401).json({ error: "Session identity signature has expired or is corrupt." });
  }
});

module.exports = router;

// Client Identity Sync Script: Secures views and mounts session profiles on dashboard assets.

let currentUserSession = null;

async function checkActiveSession() {
  try {
    const token = localStorage.getItem('urivo_access_token');
    
    if (!token) {
      window.location.href = '/login';
      return;
    }

    const response = await fetch('/api/auth/verify-session', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ token })
    });

    if (!response.ok) {
      throw new Error("Expired security identification token sequence.");
    }

    const sessionData = await response.json();
    currentUserSession = sessionData.user;

    // Synchronize UI Text Layers with runtime system metadata
    const merchantEmailNode = document.getElementById('merchantEmail');
    const creditDisplayNode = document.getElementById('creditDisplay');

    if (merchantEmailNode) merchantEmailNode.innerText = currentUserSession.email;
    if (creditDisplayNode) {
      creditDisplayNode.innerHTML = `
        ${currentUserSession.credits_remaining} <span class="text-xs font-sans tracking-wide text-urivo-ivory/60">Credits</span>
      `;
    }

  } catch (error) {
    console.error("Security Subsystem Exception Core:", error);
    localStorage.removeItem('urivo_access_token');
    window.location.href = '/login';
  }
}

function terminateActiveSession() {
  localStorage.removeItem('urivo_access_token');
  window.location.href = '/login?status=disconnected';
}

document.addEventListener('DOMContentLoaded', () => {
  if (window.location.pathname === '/dashboard') {
    checkActiveSession();
  }
});

app.use(parseSubdomain);
app.use(handleSubdomainRouting);

// Mount Subsystem Application Routing Gates
app.use(authRouter);
app.use(stripeCheckoutRouter);

// Core Gateway Presentation Router Loop
app.get('/', (req, res) => {
  if (req.isSubdomainStore) {
    const { store_name, theme_config } = req.currentStore;
    const products = req.currentProducts;

    return res.send(`
      <!DOCTYPE html>
      <html>
        <head>
          <title>${store_name}</title>
          <script src="https://tailwindcss.com"></script>
        </head>
        <body style="background: ${theme_config.palette.primaryBackground || '#EFEAD8'}; color: ${theme_config.palette.secondaryStructure || '#0B2416'}; font-family: sans-serif;" class="p-12 min-h-screen">
          <div class="max-w-4xl mx-auto space-y-12">
            <header class="border-b pb-8">
              <h1 class="text-5xl font-normal tracking-tight mb-2">${store_name}</h1>
              <p class="text-lg opacity-70 italic">${theme_config.tagline || ''}</p>
            </header>
            <main class="space-y-6">
              <div class="grid grid-cols-1 md:grid-cols-2 gap-8">
                ${products.map(p => `
                  <div class="p-6 border border-current/10 bg-white/10 flex flex-col justify-between">
                    <div>
                      <h3 class="text-xl font-medium mb-2">${p.title}</h3>
                      <p class="text-sm opacity-80 mb-4">${p.description}</p>
                    </div>
                    <div class="flex justify-between items-center mt-4">
                      <span class="text-lg font-bold">€${p.price_eur}</span>
                      <button style="background: ${theme_config.palette.accentConversion || '#C69B3C'}; color: #0B2416;" class="px-4 py-2 font-medium text-sm secure-purchase-trigger">Purchase</button>
                    </div>
                  </div>
                `).join('')}
              </div>
            </main>
          </div>
        </body>
      </html>
    `);
  }

  res.sendFile(path.join(__dirname, 'public', 'index.html'));
});

// Explicit Static View Endpoints
app.get('/dashboard', (req, res) => res.sendFile(path.join(__dirname, 'public', 'dashboard.html')));
app.get('/login', (req, res) => res.sendFile(path.join(__dirname, 'public', 'login.html')));

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`[PRODUCTION] Urivo Core Kernel active on port ${PORT}`));
