## SECTION 1: SCALABILITY & INFRASTRUCTURE

### 1.1 Rate Limiting Middleware (`lib/ratelimit.ts`)
Implementiere mit `@upstash/ratelimit` und `@upstash/redis` Schutz gegen API-Missbrauch:

```typescript
import { Ratelimit } from "@upstash/ratelimit";
import { Redis } from "@upstash/redis";

export const aiRatelimit = new Ratelimit({
  redis: Redis.fromEnv(),
  limiter: Ratelimit.slidingWindow(10, "10 s"),
  analytics: true,
  prefix: "@upstash/ratelimit/ai",
});

export async function checkRateLimit(userId: string) {
  const { success, limit, reset, remaining } = await aiRatelimit.limit(userId);

  if (!success) {
    const retryAfter = Math.ceil((reset - Date.now()) / 1000);
    throw new Error(`RATE_LIMIT_EXCEEDED:${retryAfter}`);
  }

  return { limit, remaining };
}
