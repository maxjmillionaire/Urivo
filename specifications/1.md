CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Drop existing tables to ensure clean rebuild state
DROP TABLE IF EXISTS public.credit_ledger CASCADE;
DROP TABLE IF EXISTS public.products CASCADE;
DROP TABLE IF EXISTS public.stores CASCADE;
DROP TABLE IF EXISTS public.profiles CASCADE;

-- 1. PROFILES / USERS TABLE (Linked natively to Supabase Auth)
CREATE TABLE public.profiles (
    id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    email TEXT UNIQUE NOT NULL,
    credits_remaining INT DEFAULT 15 NOT NULL CHECK (credits_remaining >= 0),
    stripe_customer_id TEXT,
    subscription_status TEXT DEFAULT 'trial' NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL
);

-- 2. STORES TABLE (Multi-Tenant Core)
CREATE TABLE public.stores (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    profile_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
    store_name TEXT NOT NULL CHECK (char_length(store_name) >= 2),
    subdomain TEXT UNIQUE NOT NULL CHECK (subdomain ~* '^[a-z0-9-]+$'),
    custom_domain TEXT UNIQUE CHECK (custom_domain ~* '^([a-z0-9]+(-[a-z0-9]+)*\.)+[a-z]{2,}$'),
    theme_config JSONB NOT NULL DEFAULT '{}'::jsonb,
    is_active BOOLEAN DEFAULT true NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL
);

-- 3. PRODUCTS TABLE (Dynamic Catalog Inventory)
CREATE TABLE public.products (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    store_id UUID REFERENCES public.stores(id) ON DELETE CASCADE NOT NULL,
    title TEXT NOT NULL CHECK (char_length(title) >= 1),
    description TEXT NOT NULL,
    price_eur NUMERIC(10, 2) NOT NULL CHECK (price_eur > 0.00),
    image_url TEXT,
    inventory_count INT DEFAULT 100 NOT NULL CHECK (inventory_count >= 0),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL
);

-- 4. CREDIT TRANSACTION LEDGER (Financial Auditing Compliance)
CREATE TABLE public.credit_ledger (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
    amount_spent INT NOT NULL CHECK (amount_spent > 0),
    action_description TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL
);

-- Indexes for rapid production database querying
CREATE INDEX idx_stores_subdomain ON public.stores(subdomain);
CREATE INDEX idx_products_store_id ON public.products(store_id);

-- Profile Sync Hook: Automatically create Profile row on Supabase Auth Signup
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, email)
  VALUES (new.id, new.email);
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

-- ATOMIC DATABASE TRANSACTION FUNCTION (Prevents credit loss on generation errors)
CREATE OR REPLACE FUNCTION generate_store_atomic_transaction(
  p_profile_id UUID,
  p_store_name TEXT,
  p_subdomain TEXT,
  p_theme_config JSONB,
  p_products JSONB,
  p_credit_cost INT
) RETURNS JSONB AS $$
DECLARE
  v_store_id UUID;
  v_credits_remaining INT;
  v_result JSONB;
BEGIN
  -- 1. Row Lock Profile Table to avoid race conditions
  SELECT credits_remaining INTO v_credits_remaining
  FROM public.profiles
  WHERE id = p_profile_id
  FOR UPDATE;

  -- 2. Validate balance again at DB level
  IF v_credits_remaining < p_credit_cost THEN
    RAISE EXCEPTION 'INSUFFICIENT_CREDITS';
  END IF;

  -- 3. Deduct active credits
  UPDATE public.profiles
  SET credits_remaining = credits_remaining - p_credit_cost
  WHERE id = p_profile_id;

  -- 4. Commit storefront row
  INSERT INTO public.stores (profile_id, store_name, subdomain, theme_config)
  VALUES (p_profile_id, p_store_name, p_subdomain, p_theme_config)
  RETURNING id INTO v_store_id;

  -- 5. Build and populate multi-product loop directly from JSONB Array payload
  INSERT INTO public.products (store_id, title, description, price_eur, inventory_count)
  SELECT 
    v_store_id,
    (elem->>'title')::TEXT,
    (elem->>'description')::TEXT,
    (elem->>'price_eur')::NUMERIC(10,2),
    COALESCE((elem->>'inventory_count')::INT, 100)
  FROM jsonb_array_elements(p_products) AS elem;

  -- 6. Log transaction ledger
  INSERT INTO public.credit_ledger (user_id, amount_spent, action_description)
  VALUES (p_profile_id, p_credit_cost, 'Dynamic storefront creation: ' || p_subdomain);

  -- Assemble dynamic response payload
  v_result := jsonb_build_object(
    'success', true,
    'store_id', v_store_id,
    'credits_remaining', v_credits_remaining - p_credit_cost
  );
  
  RETURN v_result;
EXCEPTION
  WHEN OTHERS THEN
    -- Completely reverses all transactions, releases locks, and returns error strings
    RAISE EXCEPTION '%', SQLERRM;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

### FILE 4: middleware/subdomain.js
// Robust URL Parsing: Resolves subdomains correctly on localhost, production, and staging.

function parseSubdomain(req, res, next) {
  const host = req.headers.host || '';
  const rootDomain = process.env.ROOT_DOMAIN || 'urivo.ai'; // e.g., 'urivo.ai' or 'localhost:3000'
  
  // Normalize host input (strip ports, convert to lowercase)
  const hostName = host.split(':')[0].toLowerCase().trim();
  const rootName = rootDomain.split(':')[0].toLowerCase().trim();

  // If host name matches root domain exactly, we are on the primary platform lander
  if (hostName === rootName) {
    req.subdomain = null;
    return next();
  }

  // Identify subdomain segments relative to root domain string
  if (hostName.endsWith(rootName)) {
    const subdomainPart = hostName.slice(0, -rootName.length - 1); // Subtract length and trailing dot
    
    if (subdomainPart && subdomainPart !== 'www') {
      // Clean and isolate subdomain tag
      req.subdomain = subdomainPart;
      return next();
    }
  }

  req.subdomain = null;
  next();
}

module.exports = { parseSubdomain };

### FILE 5: middleware/router.js
// Tenant-Isolated Subdomain Loader: Safely reads database configuration and blocks system exploits.

const { supabase } = require('../config/supabase');

async function handleSubdomainRouting(req, res, next) {
  // If no subdomain is present on request context, bypass to primary lander immediately
  if (!req.subdomain) {
    req.isSubdomainStore = false;
    return next();
  }

  // Security Check: Block subdomain strings that fail alphanumeric structure rules
  const subdomainRegex = /^[a-z0-9-]+$/;
  if (!subdomainRegex.test(req.subdomain)) {
    return res.status(400).send("Invalid subdomain configuration format.");
  }

  try {
    // 1. Fetch store records matching active subdomain directly
    const { data: store, error: storeError } = await supabase
      .from('stores')
      .select('*')
      .eq('subdomain', req.subdomain)
      .eq('is_active', true)
      .single();

    if (storeError || !store) {
      return res.status(404).send("The requested storefront does not exist or is currently deactivated.");
    }

    // 2. Fetch associated catalog items matching target store ID
    const { data: products, error: productError } = await supabase
      .from('products')
      .select('*')
      .eq('store_id', store.id);

    if (productError) throw productError;

    // 3. Mount payload securely onto Request context
    req.currentStore = store;
    req.currentProducts = products || [];
    req.isSubdomainStore = true;
    
    next();

  } catch (error) {
    console.error("Critical Subdomain Routing Error:", error);
    return res.status(500).send("System failed to process the store presentation layer.");
  }
}

module.exports = { handleSubdomainRouting };

### FILE 6: middleware/credits.js
// Verification Middleware: Ensures the user has a sufficient balance before initiating the expensive AI process.

const { supabase } = require('../config/supabase');

function verifyCredits(creditCost) {
  return async (req, res, next) => {
    const { userId } = req.body;

    if (!userId) {
      return res.status(400).json({ error: "Required parameter 'userId' is missing." });
    }

    try {
      // Look up balance on profiles table
      const { data: profile, error: profileError } = await supabase
        .from('profiles')
        .select('credits_remaining')
        .eq('id', userId)
        .single();

      if (profileError || !profile) {
        return res.status(404).json({ error: "User profile record not found." });
      }

      if (profile.credits_remaining < creditCost) {
        return res.status(402).json({ 
          error: "INSUFFICIENT_CREDITS", 
          creditsRemaining: profile.credits_remaining,
          required: creditCost,
          message: "Generation requires additional credits. Please upgrade your active subscription package."
        });
      }

      // Safe to pass execution flow forward
      next();

    } catch (error) {
      console.error("Database Authorization Check Failure:", error);
      return res.status(500).json({ error: "Internal security authorization failure." });
    }
  };
}

module.exports = { verifyCredits };
